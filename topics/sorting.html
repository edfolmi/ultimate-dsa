Organizing data efficiently

Sorting Algorithms Overview

Sorting is arranging data in a particular order (ascending or descending). Different algorithms have different trade-offs between time complexity, space complexity, and stability.

Comparison of Sorting Algorithms

Algorithm Best Average Worst Space Stable
Bubble Sort O(n) O(n²) O(n²) O(1) Yes
Selection Sort O(n²) O(n²) O(n²) O(1) No
Insertion Sort O(n) O(n²) O(n²) O(1) Yes
Merge Sort O(n log n) O(n log n) O(n log n) O(n) Yes
Quick Sort O(n log n) O(n log n) O(n²) O(log n) No

Implementations

// Bubble Sort
function bubbleSort(arr) {
    const n = arr.length;
    for (let i = 0; i < n - 1; i++) {
        let swapped = false;
        for (let j = 0; j < n - i - 1; j++) {
            if (arr[j] > arr[j + 1]) {
                [arr[j], arr[j + 1]] = [arr[j + 1], arr[j]];
                swapped = true;
            }
        }
        if (!swapped) break; // Optimization
    }
    return arr;
}

// Quick Sort
function quickSort(arr, low = 0, high = arr.length - 1) {
    if (low < high) {
        const pi = partition(arr, low, high);
        quickSort(arr, low, pi - 1);
        quickSort(arr, pi + 1, high);
    }
    return arr;
}

function partition(arr, low, high) {
    const pivot = arr[high];
    let i = low - 1;
    
    for (let j = low; j < high; j++) {
        if (arr[j] < pivot) {
            i++;
            [arr[i], arr[j]] = [arr[j], arr[i]];
        }
    }
    
    [arr[i + 1], arr[high]] = [arr[high], arr[i + 1]];
    return i + 1;
}

// Merge Sort
function mergeSort(arr) {
    if (arr.length <= 1) return arr;
    
    const mid = Math.floor(arr.length / 2);
    const left = mergeSort(arr.slice(0, mid));
    const right = mergeSort(arr.slice(mid));
    
    return merge(left, right);
}

function merge(left, right) {
    const result = [];
    let i = 0, j = 0;
    
    while (i < left.length && j < right.length) {
        if (left[i] < right[j]) {
            result.push(left[i++]);
        } else {
            result.push(right[j++]);
        }
    }
    
    return result.concat(left.slice(i)).concat(right.slice(j));
}

Comparing the strategies

  • Bubble sort repeatedly swaps adjacent out-of-order pairs; the swapped flag short-circuits when the array is already sorted.
  • Quick sort partitions around a pivot, then sorts the subarrays recursively—all in-place.
  • Merge sort splits the array, sorts each half, and combines them using the merge helper.
  • Seeing these side by side highlights the trade-offs between simplicity, extra space, and practical speed.

💻 Practice Problems

1. Sort Colors (Dutch National Flag) Medium

Sort an array with values 0, 1, and 2 in-place. 

function sortColors(nums) {
    let low = 0, mid = 0, high = nums.length - 1;
    
    while (mid <= high) {
        if (nums[mid] === 0) {
            [nums[low], nums[mid]] = [nums[mid], nums[low]];
            low++;
            mid++;
        } else if (nums[mid] === 1) {
            mid++;
        } else {
            [nums[mid], nums[high]] = [nums[high], nums[mid]];
            high--;
        }
    }
}

Dutch National Flag intuition

  • low tracks the boundary between zeros and ones, mid scans the array, and high marks the boundary for twos.
  • Swapping with low when we see a zero expands the left partition.
  • Swapping with high when we see a two shrinks the right partition, but we re-check the swapped value by not advancing mid.
  • Values of one already belong in the middle partition, so we just advance mid.
Time: O(n) | Space: O(1)

2. Merge Intervals Medium

Merge overlapping intervals. 

function merge(intervals) {
    if (intervals.length === 0) return [];
    
    // Sort by start time
    intervals.sort((a, b) => a[0] - b[0]);
    
    const result = [intervals[0]];
    
    for (let i = 1; i < intervals.length; i++) {
        const last = result[result.length - 1];
        const current = intervals[i];
        
        if (current[0] <= last[1]) {
            // Overlapping - merge
            last[1] = Math.max(last[1], current[1]);
        } else {
            // Non-overlapping - add new interval
            result.push(current);
        }
    }
    
    return result;
}

Reasoning about overlaps

  • Sorting by start time guarantees that overlapping intervals appear consecutively.
  • result always stores disjoint intervals; comparing with last tells us whether to merge or append.
  • Merging simply extends the end time to the maximum of the two intervals.
  • If there is no overlap we push a new entry, preserving the chronological order.
Time: O(n log n) | Space: O(n)

📚 Key Takeaways

  • O(n²) algorithms: Bubble, Selection, Insertion - simple but slow
  • O(n log n) algorithms: Merge, Quick, Heap - efficient for large data
  • Quick sort is often fastest in practice despite O(n²) worst case
  • Merge sort guarantees O(n log n) but uses O(n) extra space
  • Consider stability requirement when choosing algorithm
