Common String Problems

// Check if palindrome
function isPalindrome(s) {
    s = s.toLowerCase().replace(/[^a-z0-9]/g, '');
    let left = 0, right = s.length - 1;
    
    while (left < right) {
        if (s[left] !== s[right]) return false;
        left++;
        right--;
    }
    
    return true;
}

// Longest Palindromic Substring
function longestPalindrome(s) {
    if (s.length < 2) return s;
    
    let start = 0, maxLen = 1;
    
    function expandAroundCenter(left, right) {
        while (left >= 0 && right < s.length && s[left] === s[right]) {
            const len = right - left + 1;
            if (len > maxLen) {
                start = left;
                maxLen = len;
            }
            left--;
            right++;
        }
    }
    
    for (let i = 0; i < s.length; i++) {
        expandAroundCenter(i, i);     // Odd length
        expandAroundCenter(i, i + 1); // Even length
    }
    
    return s.substring(start, start + maxLen);
}

What each snippet shows

  • isPalindrome normalizes the string, then uses two pointers moving toward the center to compare characters.
  • longestPalindrome expands around every possible center (single character or between characters) to find the longest symmetric stretch.
  • Tracking start and maxLen lets us extract the substring at the end without storing intermediates.
  • These techniques illustrate the sliding window/two-pointer patterns used across many string problems.

💻 Practice Problems

1. Longest Substring Without Repeating Characters Medium

function lengthOfLongestSubstring(s) {
    const seen = new Map();
    let left = 0, maxLen = 0;
    
    for (let right = 0; right < s.length; right++) {
        if (seen.has(s[right])) {
            left = Math.max(left, seen.get(s[right]) + 1);
        }
        
        seen.set(s[right], right);
        maxLen = Math.max(maxLen, right - left + 1);
    }
    
    return maxLen;
}

Sliding window reasoning

  • right expands the window one character at a time.
  • The seen map remembers the most recent index of each character.
  • If we repeat a character inside the window, we move left to one past its previous index to restore uniqueness.
  • maxLen stores the best window length encountered as we scan.

📚 Key Takeaways

  • Two-pointer technique works well for strings
  • Sliding window for substring problems
  • KMP for efficient pattern matching
